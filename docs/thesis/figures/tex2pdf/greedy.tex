\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{arrows,shapes,positioning}

\begin{document}
\pagestyle{empty}
\begin{tikzpicture}[
  box/.style={rectangle, draw, thick, minimum width=6cm, minimum height=0.8cm, text centered, rounded corners, fill=white, font=\small},
  decision/.style={diamond, draw, thick, aspect=2.5, minimum width=4cm, text centered, fill=white, font=\small},
  arrow/.style={thick,->,>=stealth},
  startstop/.style={rectangle, rounded corners, draw, thick, minimum width=4cm, minimum height=0.8cm, text centered, fill=white, font=\small}]

% 主流程节点 - 垂直对齐
\node[startstop] (start) at (0,0) {开始};
\node[box] (input) at (0,-1.5) {输入: 轨迹集合 $\mathcal{T}$、检测框集合 $\mathcal{D}$、\\阈值 $\tau$}; % 添加换行
\node[box] (cost) at (0,-3) {构建代价矩阵 $\mathbf{C}$};
\node[box] (cost_detail) at (0,-4.5) {$c_{ij} = \lambda \cdot d_{motion}(i,j)$\\$+ (1-\lambda) \cdot d_{appearance}(i,j)$}; % 公式分两行
\node[box] (flatten) at (0,-6) {将代价矩阵 $\mathbf{C}$ 展平\\并按代价值升序排序}; % 添加换行
\node[box] (init) at (0,-7.5) {初始化匹配集合 $\mathcal{M} = \emptyset$};

% 循环部分
\node[box] (iterate) at (0,-9) {按代价值从小到大获取下一对$(i,j)$};
\node[decision] (pairs_check) at (0,-11) {还有未处理的\\$(i,j)$对?};
\node[decision] (check) at (0,-13.5) {$c_{ij} \leq \tau$ 且\\$T_i$, $D_j$ 均未匹配?};
\node[box] (match) at (0,-16) {添加匹配对$(T_i, D_j)$至结果集 $\mathcal{M}$\\标记 $T_i$ 和 $D_j$ 为已匹配}; % 已有换行

% 结果输出
\node[box] (output) at (0,-18) {输出: 匹配集合 $\mathcal{M}$、\\未匹配轨迹 $\mathcal{T}_{unmatched}$\\未匹配检测框 $\mathcal{D}_{unmatched}$}; % 分成三行
\node[startstop] (end) at (0,-19.5) {结束};

% 主流程箭头 - 垂直连接
\draw[arrow] (start) -- (input);
\draw[arrow] (input) -- (cost);
\draw[arrow] (cost) -- (cost_detail);
\draw[arrow] (cost_detail) -- (flatten);
\draw[arrow] (flatten) -- (init);
\draw[arrow] (init) -- (iterate);
\draw[arrow] (iterate) -- (pairs_check);

% 决策箭头
\draw[arrow] (pairs_check) -- node[right] {是} (check);
\draw[arrow] (check) -- node[right] {是} (match);

% 循环返回路径 - 增加水平偏移量避免重叠
\draw[arrow] (check) -| node[above, near start] {否} + (-6,0) |- (iterate); % 从-4增加到-6
\draw[arrow] (match) -| + (-6,0) |- (iterate); % 从-4增加到-6

% 流程结束路径 - 增加水平偏移量
\draw[arrow] (pairs_check) -| node[above] {否} + (6,0) |- (output); % 从4增加到6
\draw[arrow] (output) -- (end);

\end{tikzpicture}
\end{document}
