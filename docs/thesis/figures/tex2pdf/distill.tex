\begin{tikzpicture}[

    \draw[->] (input) -- (teacher);
    \draw[->] (input) -- (student);
    \draw[->] (teacher) -- (t_out);
    \draw[->] (student) -- (s_out);
    \draw[->] (t_out) -- (loss);
    \draw[->] (s_out) -- (loss);
    \draw[->] (loss) -- (opt);
