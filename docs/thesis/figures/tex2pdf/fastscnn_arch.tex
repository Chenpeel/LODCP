\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{ctex}
\usetikzlibrary{shapes,arrows,positioning,calc}

\begin{document}
\begin{tikzpicture}[
    node distance=1.5cm and 2cm,  % 垂直和水平间距
    module/.style={
        rectangle,
        draw=blue!50,
        fill=blue!10,
        thick,
        minimum width=3cm,
        minimum height=1cm,
        align=center,
        font=\small
    },
    branch/.style={
        rectangle,
        draw=green!50,
        fill=green!10,
        thick,
        rounded corners=5pt,
        minimum width=3cm,
        minimum height=1.2cm,
        align=center,
        font=\small
    },
    fusion/.style={
        diamond,
        draw=red!60,
        fill=red!15,
        thick,
        aspect=1.5,
        minimum width=2cm,
        align=center,
        font=\small
    },
    arrow/.style={
        ->,
        >=stealth',
        thick,
        draw=black!70
    },
    branchlabel/.style={
        font=\small\bfseries,
        above=0.3cm
    }
]

% 输入节点（顶部居中）
\node (input) [module] {输入图像\\512×512×3};

% 左侧全局特征分支
\node (global1) [module, below left=of input] {MBConv块\\步长2};
\node (global2) [module, below=of global1] {MBConv块\\步长2};
\node (ppm) [module, below=of global2] {金字塔池化\\ (PPM)};
\node (global_feat) [branch, below=of ppm] {全局特征\\16×16×128};

% 右侧局部特征分支
\node (down1) [module, below right=of input] {3×3卷积\\步长2};
\node (down2) [module, below=of down1] {深度可分离卷积\\步长2};
\node (down3) [module, below=of down2] {深度可分离卷积\\步长1};
\node (local_feat) [branch, below=of down3] {局部特征\\128×128×64};

% 特征融合（与左右分支末端平级）
\node (fusion) [fusion, right=of global_feat, left=of local_feat] {特征融合\\注意力机制};

% 输出分支（下方居中）
\node (decoder) [module, below=2cm of fusion] {上采样+1×1卷积};
\node (output) [module, below=of decoder] {输出分割图\\512×512×C};

% 连接箭头
% 左侧分支
\draw [arrow] (input) -| (global1);
\draw [arrow] (global1) -- (global2);
\draw [arrow] (global2) -- (ppm);
\draw [arrow] (ppm) -- (global_feat);
\draw [arrow] (global_feat.east) -- (fusion.west);

% 右侧分支
\draw [arrow] (input) -| (down1);
\draw [arrow] (down1) -- (down2);
\draw [arrow] (down2) -- (down3);
\draw [arrow] (down3) -- (local_feat);
\draw [arrow] (local_feat.west) -- (fusion.east);

% 垂直输出
\draw [arrow] (fusion) -- (decoder);
\draw [arrow] (decoder) -- (output);

% 分支标注
\node [branchlabel] at ($(global1.north)+(0,0.3cm)$) {全局特征分支};
\node [branchlabel] at ($(down1.north)+(0,0.3cm)$) {学习下采样分支};

\end{tikzpicture}
